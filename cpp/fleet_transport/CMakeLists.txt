cmake_minimum_required(VERSION 3.16)
project(chronus_fleet_transport LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
add_library(chronus_fleet_transport SHARED transport.cpp)
target_compile_features(chronus_fleet_transport PRIVATE cxx_std_17)
target_link_libraries(chronus_fleet_transport PRIVATE PkgConfig::CURL)
set_target_properties(chronus_fleet_transport PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(MSVC)
    target_compile_options(chronus_fleet_transport PRIVATE /W4)
else()
    target_compile_options(chronus_fleet_transport PRIVATE -Wall -Wextra -Wpedantic)
endif()
