This is an evidence framework, not an AGI declaration.  It distinguishes
implemented surfaces, measured capability, evidence quality, and comparable
longitudinal growth under a fixed schema.

_DEFINITION_BY_VALUE={definition.attribute.value:definition for definition in DEFINITIONS}
_SCHEMA_PAYLOAD=[{"attribute":d.attribute.value,"metrics":d.metrics,"modules":d.module_evidence} for d in DEFINITIONS]
SCHEMA_FINGERPRINT=hashlib.sha256(json.dumps(_SCHEMA_PAYLOAD,sort_keys=True).encode()).hexdigest()


@dataclass(frozen=True)
class AttributeMeasurement:
    attribute:str
    score:float
    coverage:float
    evidence_quality:float
    confidence:float
    observed_metrics:dict[str,float]
    missing_metrics:tuple[str,...]
    module_evidence:tuple[str,...]


@dataclass(frozen=True)
class CapabilitySnapshot:
    snapshot_id:str
    label:str
    timestamp:float
    schema_version:str
    schema_fingerprint:str
    measurements:dict[str,AttributeMeasurement]
    aggregate_score:float
    aggregate_confidence:float
    evidence_coverage:float
    bottleneck_score:float
    necessary_attribute_gate_passed:bool
    metadata:dict[str,Any]

    def to_dict(self)->dict[str,Any]:
        result=asdict(self)
        return result

    @classmethod
    def from_dict(cls,data:Mapping[str,Any])->"CapabilitySnapshot":
        measurements={name:AttributeMeasurement(**value) for name,value in data["measurements"].items()}
        return cls(snapshot_id=str(data["snapshot_id"]),label=str(data["label"]),timestamp=float(data["timestamp"]),
            schema_version=str(data["schema_version"]),schema_fingerprint=str(data["schema_fingerprint"]),
            measurements=measurements,aggregate_score=float(data["aggregate_score"]),
            aggregate_confidence=float(data["aggregate_confidence"]),evidence_coverage=float(data["evidence_coverage"]),
            bottleneck_score=float(data["bottleneck_score"]),
            necessary_attribute_gate_passed=bool(data["necessary_attribute_gate_passed"]),
            metadata=dict(data.get("metadata",{})))


@dataclass(frozen=True)
class GrowthReport:
    baseline_id:str
    latest_id:str
    snapshots_compared:int
    aggregate_delta:float
    confidence_delta:float
    coverage_delta:float
    attribute_deltas:dict[str,float]
    attribute_slopes:dict[str,float]
    improving:tuple[str,...]
    stable:tuple[str,...]
    regressing:tuple[str,...]


class AGIAttributeEvaluator:
    """Evaluate and persist comparable snapshots under a fixed metric schema."""

    def __init__(self,state_path:str|None=None,attribute_threshold:float=.7,
                 confidence_threshold:float=.7,min_growth_delta:float=.01):
        for value in (attribute_threshold,confidence_threshold):
            if not 0<=value<=1:raise ValueError("thresholds must be in [0,1]")
        if min_growth_delta<0:raise ValueError("min_growth_delta must be nonnegative")
        self.state_path=state_path;self.attribute_threshold=float(attribute_threshold)
        self.confidence_threshold=float(confidence_threshold);self.min_growth_delta=float(min_growth_delta)
        self.snapshots:list[CapabilitySnapshot]=[]
        if state_path and Path(state_path).exists():self._load()

    @staticmethod
    def _validate_score(value:Any,name:str)->float:
        score=float(value)
        if not math.isfinite(score) or not 0<=score<=1:raise ValueError(f"{name} must be finite and in [0,1]")
        return score

    def evaluate(self,metrics:Mapping[str,Mapping[str,float]],*,label:str,
                 evidence_quality:Mapping[str,float]|None=None,metadata:Mapping[str,Any]|None=None,
                 persist:bool=True)->CapabilitySnapshot:
        if not label:raise ValueError("label must not be empty")
        unknown_attributes=set(metrics)-set(_DEFINITION_BY_VALUE)
        if unknown_attributes:raise ValueError(f"unknown attributes: {sorted(unknown_attributes)}")
        quality_map=dict(evidence_quality or {});measurements={}
        for definition in DEFINITIONS:
            name=definition.attribute.value;provided=dict(metrics.get(name,{}))
            unknown_metrics=set(provided)-set(definition.metrics)
            if unknown_metrics:raise ValueError(f"unknown metrics for {name}: {sorted(unknown_metrics)}")
            observed={metric:self._validate_score(value,f"{name}.{metric}") for metric,value in provided.items()}
            # Fixed equal weights; missing evidence contributes zero rather than disappearing.
            score=sum(observed.get(metric,0.0) for metric in definition.metrics)/len(definition.metrics)
            coverage=len(observed)/len(definition.metrics)
            quality=self._validate_score(quality_map.get(name,1.0),f"{name}.evidence_quality")
            confidence=coverage*quality
            missing=tuple(metric for metric in definition.metrics if metric not in observed)
            measurements[name]=AttributeMeasurement(name,score,coverage,quality,confidence,observed,missing,
                                                     definition.module_evidence)
        values=list(measurements.values());aggregate=sum(item.score for item in values)/len(values)
        aggregate_confidence=sum(item.confidence for item in values)/len(values)
        coverage=sum(item.coverage for item in values)/len(values);bottleneck=min(item.score for item in values)
        gate=all(item.score>=self.attribute_threshold and item.confidence>=self.confidence_threshold
                 for item in values)
        snapshot=CapabilitySnapshot(str(uuid.uuid4()),label,time.time(),SCHEMA_VERSION,SCHEMA_FINGERPRINT,
            measurements,aggregate,aggregate_confidence,coverage,bottleneck,gate,dict(metadata or {}))
        self.snapshots.append(snapshot)
        if persist:self._checkpoint()
        return snapshot

    def growth_report(self,baseline_index:int=0,latest_index:int=-1)->GrowthReport:
        if len(self.snapshots)<2:raise ValueError("at least two snapshots are required")
        baseline=self.snapshots[baseline_index];latest=self.snapshots[latest_index]
        if baseline.schema_fingerprint!=latest.schema_fingerprint:raise ValueError("snapshots use different schemas")
        deltas={name:latest.measurements[name].score-baseline.measurements[name].score for name in _DEFINITION_BY_VALUE}
        slopes={name:self._slope([snapshot.measurements[name].score for snapshot in self.snapshots])
                for name in _DEFINITION_BY_VALUE}
        improving=tuple(name for name,value in deltas.items() if value>self.min_growth_delta)
        regressing=tuple(name for name,value in deltas.items() if value<-self.min_growth_delta)
        stable=tuple(name for name,value in deltas.items() if abs(value)<=self.min_growth_delta)
        return GrowthReport(baseline.snapshot_id,latest.snapshot_id,len(self.snapshots),
            latest.aggregate_score-baseline.aggregate_score,
            latest.aggregate_confidence-baseline.aggregate_confidence,
            latest.evidence_coverage-baseline.evidence_coverage,deltas,slopes,improving,stable,regressing)

    @staticmethod
    def _slope(values:list[float])->float:
        if len(values)<2:return 0.0
        x_mean=(len(values)-1)/2;y_mean=sum(values)/len(values)
        denominator=sum((index-x_mean)**2 for index in range(len(values)))
        return sum((index-x_mean)*(value-y_mean) for index,value in enumerate(values))/denominator if denominator else 0.0

    def _checkpoint(self)->None:
        if not self.state_path:return
        payload={"schema_version":SCHEMA_VERSION,"schema_fingerprint":SCHEMA_FINGERPRINT,
                 "snapshots":[snapshot.to_dict() for snapshot in self.snapshots]}
        verified_atomic_json_write(Path(self.state_path).resolve(),payload)

    def _load(self)->None:
        with open(self.state_path,encoding="utf-8") as handle:payload=json.load(handle)
        if payload.get("schema_fingerprint")!=SCHEMA_FINGERPRINT:
            raise ValueError("persisted AGI attribute schema is incompatible")
        self.snapshots=[CapabilitySnapshot.from_dict(item) for item in payload.get("snapshots",[])]

    def clear(self,persist:bool=True)->None:
        self.snapshots.clear()
        if persist:self._checkpoint()
