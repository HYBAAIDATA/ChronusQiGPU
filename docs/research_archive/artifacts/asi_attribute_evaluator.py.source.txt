An ASI gate requires complete internal evidence, high-quality measurements,
an independently supplied reference baseline, and a passed AGI prerequisite.
Module names and self-reported scores can never satisfy the gate alone.

_BY_VALUE={d.attribute.value:d for d in DEFINITIONS}
SCHEMA_FINGERPRINT=hashlib.sha256(json.dumps([asdict(d) for d in DEFINITIONS],sort_keys=True,default=str).encode()).hexdigest()

@dataclass(frozen=True)
class ASIAttributeMeasurement:
    attribute:str;score:float;coverage:float;evidence_quality:float;confidence:float
    observed_metrics:dict[str,float];missing_metrics:tuple[str,...]
    reference_coverage:float;reference_ratios:dict[str,float];superhuman_verified:bool
    module_evidence:tuple[str,...]

@dataclass(frozen=True)
class ASISnapshot:
    snapshot_id:str;label:str;timestamp:float;schema_version:str;schema_fingerprint:str
    measurements:dict[str,ASIAttributeMeasurement];aggregate_score:float;aggregate_confidence:float
    evidence_coverage:float;reference_coverage:float;bottleneck_score:float
    agi_prerequisite_passed:bool;asi_gate_passed:bool;metadata:dict[str,Any]
    def to_dict(self):return asdict(self)
    @classmethod
    def from_dict(cls,d):
        ms={k:ASIAttributeMeasurement(**v) for k,v in d["measurements"].items()}
        return cls(d["snapshot_id"],d["label"],float(d["timestamp"]),d["schema_version"],d["schema_fingerprint"],ms,
            float(d["aggregate_score"]),float(d["aggregate_confidence"]),float(d["evidence_coverage"]),
            float(d["reference_coverage"]),float(d["bottleneck_score"]),bool(d["agi_prerequisite_passed"]),
            bool(d["asi_gate_passed"]),dict(d.get("metadata",{})))

@dataclass(frozen=True)
class ASIGrowthReport:
    baseline_id:str;latest_id:str;snapshots_compared:int;aggregate_delta:float
    confidence_delta:float;coverage_delta:float;reference_coverage_delta:float
    attribute_deltas:dict[str,float];improving:tuple[str,...];stable:tuple[str,...];regressing:tuple[str,...]

class ASIAttributeEvaluator:
    def __init__(self,state_path:str|None=None,attribute_threshold:float=.9,confidence_threshold:float=.8,min_growth_delta:float=.01):
        if any(not 0<=v<=1 for v in (attribute_threshold,confidence_threshold)) or min_growth_delta<0:raise ValueError("invalid thresholds")
        self.state_path=state_path;self.attribute_threshold=attribute_threshold;self.confidence_threshold=confidence_threshold
        self.min_growth_delta=min_growth_delta;self.snapshots:list[ASISnapshot]=[]
        if state_path and Path(state_path).exists():self._load()
    @staticmethod
    def _unit(value:Any,name:str)->float:
        value=float(value)
        if not math.isfinite(value) or not 0<=value<=1:raise ValueError(f"{name} must be finite and in [0,1]")
        return value
    @staticmethod
    def _ratio(value:Any,name:str)->float:
        value=float(value)
        if not math.isfinite(value) or value<0:raise ValueError(f"{name} must be finite and nonnegative")
        return value
    def evaluate(self,metrics:Mapping[str,Mapping[str,float]],*,label:str,evidence_quality:Mapping[str,float]|None=None,
                 reference_ratios:Mapping[str,Mapping[str,float]]|None=None,agi_prerequisite_passed:bool=False,
                 metadata:Mapping[str,Any]|None=None,persist:bool=True)->ASISnapshot:
        if not label:raise ValueError("label must not be empty")
        unknown=set(metrics)-set(_BY_VALUE)
        if unknown:raise ValueError(f"unknown ASI attributes: {sorted(unknown)}")
        qualities=dict(evidence_quality or {});references=dict(reference_ratios or {});measurements={}
        for definition in DEFINITIONS:
            name=definition.attribute.value;provided=dict(metrics.get(name,{}));unknown_metrics=set(provided)-set(definition.metrics)
            if unknown_metrics:raise ValueError(f"unknown metrics for {name}: {sorted(unknown_metrics)}")
            observed={k:self._unit(v,f"{name}.{k}") for k,v in provided.items()};missing=tuple(k for k in definition.metrics if k not in observed)
            score=sum(observed.get(k,0) for k in definition.metrics)/len(definition.metrics);coverage=len(observed)/len(definition.metrics)
            quality=self._unit(qualities.get(name,1),f"{name}.quality");confidence=coverage*quality
            supplied=dict(references.get(name,{}));unknown_refs=set(supplied)-set(definition.reference_metrics)
            if unknown_refs:raise ValueError(f"unknown reference metrics for {name}: {sorted(unknown_refs)}")
            ratios={k:self._ratio(v,f"{name}.reference.{k}") for k,v in supplied.items()}
            ref_coverage=len(ratios)/len(definition.reference_metrics) if definition.reference_metrics else 1.0
            verified=ref_coverage==1 and all(ratios[k]>1 for k in definition.reference_metrics)
            measurements[name]=ASIAttributeMeasurement(name,score,coverage,quality,confidence,observed,missing,ref_coverage,ratios,verified,definition.module_evidence)
        values=list(measurements.values());aggregate=sum(x.score for x in values)/len(values);confidence=sum(x.confidence for x in values)/len(values)
        coverage=sum(x.coverage for x in values)/len(values);ref_coverage=sum(x.reference_coverage for x in values)/len(values);bottleneck=min(x.score for x in values)
        gate=bool(agi_prerequisite_passed and all(x.score>=self.attribute_threshold and x.confidence>=self.confidence_threshold and x.superhuman_verified for x in values))
        snapshot=ASISnapshot(str(uuid.uuid4()),label,time.time(),SCHEMA_VERSION,SCHEMA_FINGERPRINT,measurements,aggregate,confidence,coverage,ref_coverage,bottleneck,bool(agi_prerequisite_passed),gate,dict(metadata or {}))
        self.snapshots.append(snapshot)
        if persist:self._checkpoint()
        return snapshot
    def growth_report(self,baseline_index:int=0,latest_index:int=-1)->ASIGrowthReport:
        if len(self.snapshots)<2:raise ValueError("at least two snapshots are required")
        a,b=self.snapshots[baseline_index],self.snapshots[latest_index];d={k:b.measurements[k].score-a.measurements[k].score for k in _BY_VALUE}
        return ASIGrowthReport(a.snapshot_id,b.snapshot_id,len(self.snapshots),b.aggregate_score-a.aggregate_score,b.aggregate_confidence-a.aggregate_confidence,
            b.evidence_coverage-a.evidence_coverage,b.reference_coverage-a.reference_coverage,d,
            tuple(k for k,v in d.items() if v>self.min_growth_delta),tuple(k for k,v in d.items() if abs(v)<=self.min_growth_delta),tuple(k for k,v in d.items() if v<-self.min_growth_delta))
    def _checkpoint(self):
        if not self.state_path:return
        payload={"schema_version":SCHEMA_VERSION,"schema_fingerprint":SCHEMA_FINGERPRINT,"snapshots":[s.to_dict() for s in self.snapshots]}
        verified_atomic_json_write(Path(self.state_path).resolve(),payload)
    def _load(self):
        with open(self.state_path) as f:payload=json.load(f)
        if payload.get("schema_fingerprint")!=SCHEMA_FINGERPRINT:raise ValueError("persisted ASI schema is incompatible")
        self.snapshots=[ASISnapshot.from_dict(x) for x in payload.get("snapshots",[])]
