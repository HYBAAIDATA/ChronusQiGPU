cmake_minimum_required(VERSION 3.16)
project(chronus1_emulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(chronus1 src/decode.cpp src/memory.cpp src/loader.cpp src/emulator.cpp)
target_include_directories(chronus1 PUBLIC include)
if(MSVC)
 target_compile_options(chronus1 PRIVATE /fp:strict /W4)
else()
 target_compile_options(chronus1 PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
endif()
add_executable(chronus-emu src/main.cpp)
target_link_libraries(chronus-emu PRIVATE chronus1)
add_executable(chronus-emulator-tests tests/test_emulator.cpp)
target_link_libraries(chronus-emulator-tests PRIVATE chronus1)
enable_testing()
add_test(NAME chronus-emulator-tests COMMAND chronus-emulator-tests)
