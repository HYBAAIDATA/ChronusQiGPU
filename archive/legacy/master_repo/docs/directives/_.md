def rfc8785(obj: Any) -> bytes:
    """
    RFC 8785 JSON Canonicalization Scheme.
    Deterministic byte representation across machines and runtimes — required
    so that sha256(rfc8785(payload)) is reproducible by any verifier.
    """
    return json.dumps(
        obj, sort_keys=True, separators=(',', ':'), ensure_ascii=False
    ).encode('utf-8')

def sha256(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()

def sha256_raw(b: bytes) -> bytes:
    return hashlib.sha256(b).digest()

def sign_ed25519(key: Any, msg: str) -> str:
    """
    Ed25519 signing stub.  In production: cryptography.hazmat.primitives.
    Returns a deterministic 64-hex-char string for simulation purposes.
    """
    return hashlib.sha256((str(key) + msg).encode()).hexdigest()

def verify_ed25519(pub: bytes, msg: str, sig: str) -> bool:
    """Verification stub — always passes in simulation; real impl checks curve."""
    return len(sig) == 64

def now_iso8601() -> str:
    from datetime import datetime, timezone
    return datetime.now(timezone.utc).isoformat()

def fib_fast(n: int) -> Tuple[int, int]:
    """
    Fast doubling Fibonacci: returns (F(n), F(n+1)) in O(log n) integer ops.
    Used by cassini_witness to verify the Cassini identity mod 256.
    """
    if n <= 0:
        return (0, 1)
    def _fib(num: int) -> Tuple[int, int]:
        if num == 0:
            return (0, 1)
        a, b = _fib(num >> 1)
        c = a * (2 * b - a)
        d = a * a + b * b
        return (d, c + d) if (num & 1) else (c, d)
    return _fib(n)

class TensorMemoryAccelerator:
    """
    Autonomous on-die DMA unit.  Stages multi-dimensional tensor tiles directly
    from DRAM/HBM into local SM SRAM without consuming register files or warp

    """

    def __init__(self):

    def stage_tile_async(
        self,
        sram_target: bytearray,
        dram_slice:  memoryview,
        tile_shape:  Tuple[int, int]
    ):
        self.barrier_tokens += 1

        sram_target[: len(dram_slice)] = dram_slice[:]

        """Hardware sync token — confirms tile has landed in local SRAM."""
        self.barrier_tokens = 0

class PrecisionFormat(Enum):
    FP32    = 32   # Master accumulator format
    FP16    = 16   # Mixed-precision baseline
    FP8_E4M3 = 8   # Hopper-era matrix multiply
    NVFP4   =  4   # Blackwell/Rubin: extreme tokens-per-watt

class MicroScaleBlock:
    """Dynamic per-block exponent/mantissa scale for low-precision quantisation."""

    def __init__(self, scale_value: float):
        self.scale = scale_value

class TransformerMicroEngine:
    """
    Monitors activation dynamics and scales FP16 vectors into NVFP4 sub-blocks
    without numerical precision collapse.
    """

    @staticmethod
    def quantize_to_nvfp4(
        vector_fp16: List[float]
    ) -> Tuple[List[int], MicroScaleBlock]:
        max_abs = max(abs(x) for x in vector_fp16) if vector_fp16 else 1.0
        scale   = max_abs / 6.0 if max_abs > 0 else 1.0
        # Clamp to signed 4-bit boundary [-8, 7]
        quantized = [
            min(7, max(-8, int(round(val / (scale + 1e-7)))))
            for val in vector_fp16
        ]
        return quantized, MicroScaleBlock(scale)

class TensorCoreEngine:
    """
    Specialised matrix execution pipeline.
    Executes fused MMA (matrix multiply-accumulate) directly on NVFP4 / FP8 tiles
    inside the SM, accumulating into FP32.
    """

    def compute_nvfp4_mma(
        self,
        a_nvfp4:         List[int],
        scale_a:         MicroScaleBlock,
        b_nvfp4:         List[int],
        scale_b:         MicroScaleBlock,
        accumulator_fp32: List[float]
    ) -> List[float]:
        combined_scale = scale_a.scale * scale_b.scale
        dot_product    = sum(a * b for a, b in zip(a_nvfp4, b_nvfp4)) * combined_scale
        for i in range(len(accumulator_fp32)):
            accumulator_fp32[i] += dot_product
        return accumulator_fp32

class ExecutionWarp:
    """32-thread lockstep execution group with active masking for divergence."""

    def __init__(self, warp_id: int, num_lanes: int = 32):
        self.warp_id    = warp_id
        self.num_lanes  = num_lanes
        self.active_mask = [True] * num_lanes

    def apply_branch_divergence(self, predicate_flags: List[bool]):
        """Disables threads that diverge on a conditional branch."""
        self.active_mask = [
            act and pred
            for act, pred in zip(self.active_mask, predicate_flags)
        ]

class StreamingMultiprocessor:
    """
    The fundamental compute unit.  Houses:
      - 32-lane SIMD vector ALUs
      - Register files per lane
      - Warp Scheduler (dual-issue)
      - Tensor Memory Accelerator (TMA) interface
      - Tensor Core Engine (NVFP4 / FP8 MMA)
      - 256 KB unified L1/Shared SRAM
    """

    def __init__(self, sm_id: int, num_lanes: int = 32):
        self.sm_id           = sm_id
        self.num_lanes       = num_lanes
        self.l1_shared_sram  = bytearray(256 * 1024)
        self.register_file   = [{} for _ in range(num_lanes)]
        self.tma             = TensorMemoryAccelerator()
        self.tensor_core     = TensorCoreEngine()

    def execute_vector_instruction(
        self,
        warp:  ExecutionWarp,
        instr: SIMDInstruction
    ):
        for lane in range(self.num_lanes):
            if warp.active_mask[lane]:
                val_a = self.register_file[lane].get(instr.src_reg_a, 0)
                val_b = self.register_file[lane].get(instr.src_reg_b, 0)
                res   = (val_a + val_b) if instr.op_name == "ADD" else (val_a * val_b)
                self.register_file[lane][instr.dest_reg] = res

    def execute_matrix_tile_pipeline(self, memory_slice: memoryview) -> List[float]:
        """
        1. Asynchronously stages tensor tile via TMA into SRAM.
        2. Dynamically quantises to NVFP4.
        3. Executes dense Tensor Core MMA.
        """
        self.tma.stage_tile_async(
            self.l1_shared_sram, memory_slice, tile_shape=(16, 16)
        )
        self.tma.await_barrier()

        sample_w = [0.25, -0.65, 0.95, -0.10] * 4
        sample_a = [0.15,  0.35, -0.20,  0.80] * 4
        w_nvfp4, scale_w = TransformerMicroEngine.quantize_to_nvfp4(sample_w)
        a_nvfp4, scale_a = TransformerMicroEngine.quantize_to_nvfp4(sample_a)

        accumulator = [0.0] * 16
        return self.tensor_core.compute_nvfp4_mma(
            w_nvfp4, scale_w, a_nvfp4, scale_a, accumulator
        )

    Superconducting variant sits at 10–20 mK inside a dilution refrigerator,
    RF-isolated from the classical GPU rack via silicon photonics fiber links.
    """

        modality:            QubitModality = QubitModality.SUPERCONDUCTING
    ):

        self.modality        = modality
        self.ppu             = PulseProcessingUnit()

    def execute_raw_circuit(
        self,
        gate_sequence: List[Dict[str, Any]]
    ) -> List[int]:
        """
        Fires pulse waveforms per gate, then collapses wavefunctions.
        Returns classical measurement bitstrings.

        Gate shapes supported:
          Single-qubit: {"gate": "H"|"RY", "qubit": int, "theta"?: float}
          Two-qubit:    {"gate": "CNOT", "control": int, "target": int}
        """
        for op in gate_sequence:
            gate = op.get("gate", "")
            if gate == "CNOT":
                # Two-qubit gate: pulse both control and target lines
                for q in (op["control"], op["target"]):
                    self.ppu.emit_microwave_pulse(
                        q, freq_ghz=5.0, phase_rad=0.0, duration_ns=40.0
                    )
            else:
                # Single-qubit gate (H, RY, etc.)
                self.ppu.emit_microwave_pulse(
                    op["qubit"],
                    freq_ghz    = 5.0,
                    phase_rad   = op.get("theta", 0.0),
                    duration_ns = 20.0
                )
        return [0, 1, 0, 0, 1, 1, 0, 0]   # Measured bitstrings

    #  CORE: Fault-Tolerant Kernel Execution with Active Real-Time QEC    #
    # ------------------------------------------------------------------ #

    def run_fault_tolerant_kernel(
        self,
        circuit: QuantumCircuitAnsatz
    ) -> List[float]:
        """
        Executes a quantum circuit interleaved with active GPU QEC:
          1. Fire physical gate pulses on the QPU.
          2. Stream syndrome bits to GPU HBM over NVQLink (< 0.6 µs).
          3. Tensor Core decoder identifies Pauli errors.
          4. Dispatch corrective pulses back before decoherence.
          5. Return expectation values from fault-tolerant logical states.
        """
        raw_syndrome_bits = self.qpu.execute_raw_circuit(circuit.operations)
        self.nvqlink.stream_syndromes_to_gpu_hbm(
            raw_syndrome_bits, memory_dest_addr=0x8000_0000
        )
        corrections = self.qec_decoder.decode_syndromes(raw_syndrome_bits)
        self.nvqlink.dispatch_pauli_corrections(corrections, self.qpu.ppu)
        return [0.852, 0.148]

    # ------------------------------------------------------------------ #
    #  CAPABILITY 1: Quantum-Assisted Attention (SWAP-Test Overlap)        #
    # ------------------------------------------------------------------ #

    def evaluate_quantum_attention(
        self,
        q_vector: List[float],
        k_vector: List[float]
    ) -> float:
        """
        Projects Query and Key embedding vectors onto the exponential 2^N
        Hilbert state space using a quantum SWAP test, computing the exact
        inner-product fidelity as an attention score — without O(d²) classical cost.
        """
        circuit = QuantumCircuitAnsatz(num_qubits=4)
        circuit.apply_hadamard(0)
        angle_q = math.atan2(q_vector[1], q_vector[0])
        angle_k = math.atan2(k_vector[1], k_vector[0])
        circuit.apply_parameterized_ry(1, angle_q)
        circuit.apply_parameterized_ry(2, angle_k)
        circuit.apply_cnot(1, 2)
        circuit.apply_hadamard(0)

        expectation = self.run_fault_tolerant_kernel(circuit)
        # Fidelity overlap → attention score
        return max(0.0, expectation[0] - expectation[1])

    # ------------------------------------------------------------------ #
    #  CAPABILITY 2: Variational Quantum Eigensolver (VQE / QAOA)         #
    # ------------------------------------------------------------------ #

    def execute_vqe_subroutine(
        self,
        hamiltonian_terms: List[str],
        gpu_weights:       List[float]
    ) -> float:
        """
        Hybrid variational loop:
          - GPU computes analytical parameter gradients (parameter-shift rule).
          - QPU evaluates ground-state molecular or combinatorial Hamiltonians.
        Suitable for: drug discovery, logistics optimisation, graph partitioning.
        """
        circuit = QuantumCircuitAnsatz(num_qubits=4)
        for i, weight in enumerate(gpu_weights[:4]):
            circuit.apply_hadamard(i)
            circuit.apply_parameterized_ry(i, weight)
        for i in range(3):
            circuit.apply_cnot(i, i + 1)

        result            = self.run_fault_tolerant_kernel(circuit)
        energy_expectation = sum(result) * -1.414   # Rayleigh-Ritz energy estimate
        return energy_expectation

    # ------------------------------------------------------------------ #
    #  CAPABILITY 3: Quantum Neural Network (QNN) Forward Pass             #
    # ------------------------------------------------------------------ #

    def execute_qnn_forward_pass(
        self,
        input_embeddings: List[float],
        layer_depth:      int = 3
    ) -> List[float]:
        """
        Executes a multi-layer parameterised QNN ansatz.  Each layer applies:
          H gates (superposition)  →  RY rotations (data encoding)  →
          CNOT entanglement ladder  →  measure expectation values.
        GPU back-propagates gradients through the quantum layer via
        parameter-shift differentiation.
        """
        circuit = QuantumCircuitAnsatz(num_qubits=min(4, len(input_embeddings)))
        for layer in range(layer_depth):
            for q in range(circuit.num_qubits):
                circuit.apply_hadamard(q)
                angle = input_embeddings[q % len(input_embeddings)] * (layer + 1)
                circuit.apply_parameterized_ry(q, angle)
            for q in range(circuit.num_qubits - 1):
                circuit.apply_cnot(q, q + 1)

        return self.run_fault_tolerant_kernel(circuit)

    # ------------------------------------------------------------------ #
    #  CAPABILITY 4: Quantum Sampling (Random / Distribution Estimation)  #
    # ------------------------------------------------------------------ #

    def quantum_distribution_sample(self, num_qubits: int = 8) -> List[int]:
        """
        Prepares a uniform superposition across 2^N basis states and collapses
        to draw an unbiased bitstring sample in a single QPU shot.
        Used for: Monte Carlo acceleration, generative modelling, random seed
        generation immune to classical PRNG biases.
        """
        circuit = QuantumCircuitAnsatz(num_qubits=num_qubits)
        for q in range(num_qubits):
            circuit.apply_hadamard(q)
        raw = self.qpu.execute_raw_circuit(circuit.operations)
        return raw

class AcceleratedBladeNode:
    """
    Individual server blade containing:
      - Unified memory slice (zero-copy mapped)
      - 144 Streaming Multiprocessors (SM array)
      - NVLink + NVQLink interface ports
      - Local KV-cache store (HBM-resident)
    """

    def __init__(self, node_id: int, unified_memory: UnifiedPhysicalMemory):
        self.node_id    = node_id
        self.memory     = unified_memory
        self.sms        = [StreamingMultiprocessor(i) for i in range(144)]
        self.kv_storage: Dict[str, Any] = {}

    def prefill_context(self, tokens: List[int]) -> str:
        """
        COMPUTE-BOUND PHASE (high FLOP density):
        Parallel ingestion of full context window via NVFP4 Tensor Core MMA.
        Populates KV-cache matrices in HBM.
        """
        job_id = f"job_node_{self.node_id}"
        zero_copy_view = self.memory.request_zero_copy_view(
            MemoryClientPriority.TMA_ASYNC_BLOCK, addr=0x2000, length=256
        )
        for sm in self.sms[:8]:
            sm.execute_matrix_tile_pipeline(zero_copy_view)

        self.kv_storage[job_id] = {
            "token_count": len(tokens),
            "kv_matrix":   "HBM_ALLOCATED"
        }
        return job_id

    def decode_token_stream(
        self,
        job_id:     str,
        max_tokens: int
    ) -> List[int]:
        """
        MEMORY-BANDWIDTH-BOUND PHASE (high stream density):
        Autoregressively generates output tokens using NVFP4 parameter weights
        streamed from HBM one KV-row at a time.
        """
        return [100 + step for step in range(max_tokens)]

        base_tokens = decode_worker.decode_token_stream(job_id, max_tokens=16)
        output_tokens = [
            int(tok * quantum_attn_score * (qnn_output[0] if qnn_output else 1.0)) % 32_000
            for tok in base_tokens
        ]
        return output_tokens

# TIER 12: EXACT ALGEBRAIC UNIT
#          Z[φ] Ring Arithmetic, Cassini M15 Matrix & Fold Core
# =============================================================================
#
# Z[φ] is the ring of integers extended by the golden ratio φ = (1+√5)/2.
# Elements are pairs (a, b) representing a + b·φ.
# Multiplication follows φ² = φ + 1  →  (a+bφ)(c+dφ) = (ac+bd) + (ad+bc+bd)φ.
#
# The M15 matrix is a 2×2 integer matrix over Z/256Z with determinant ≡ 219 (mod 256).
# Its inverse M15_INV satisfies M15 · M15_INV ≡ I (mod 256).
# The Cassini identity F(n+1)·F(n-1) − F(n)² = (−1)^n is verified mod 256
# as a witness that the fold kernel is operating on a coherent Fibonacci lattice.
# =============================================================================

class RingElem:
    """An element a + b·φ in the ring Z[φ]."""
    def __init__(self, a: int, b: int):
        self.a = a
        self.b = b

    def __repr__(self) -> str:
        return f"Z[φ]({self.a} + {self.b}φ)"

class ExactRing:
    """
    Z[φ] arithmetic engine bound to a specific tenant signing context.
    When mod_256 is True, all coefficients are reduced modulo 256,
    making this a ring over Z/256Z[φ] suitable for byte-level fold operations.
    """

    def __init__(
        self,
        tenant_hash: str,
        mod_256:     bool,
        signing_key: Any,
        key_id:      str,
        public_key:  bytes
    ):
        self.tenant_hash = tenant_hash
        self.mod_256     = mod_256
        self.signing_key = signing_key
        self.key_id      = key_id
        self.public_key  = public_key

    def _r(self, v: int) -> int:
        return v % 256 if self.mod_256 else v

    def add(self, x: RingElem, y: RingElem) -> RingElem:
        return RingElem(self._r(x.a + y.a), self._r(x.b + y.b))

    def mul(self, x: RingElem, y: RingElem) -> RingElem:
        """
        (a + bφ)(c + dφ) = (ac + bd) + (ad + bc + bd)φ
        Uses φ² = φ + 1.
        """
        ra = self._r(x.a * y.a + x.b * y.b)
        rb = self._r(x.a * y.b + x.b * y.a + x.b * y.b)
        return RingElem(ra, rb)

    def conj(self, x: RingElem) -> RingElem:
        """Galois conjugate: (a + bφ)* = a + b(1−φ) = (a+b) − bφ."""
        return RingElem(self._r(x.a + x.b), self._r(-x.b))

    def trace(self, x: RingElem) -> int:
        """Tr(a + bφ) = x + x* = 2a + b."""
        return self._r(2 * x.a + x.b)

    def norm(self, x: RingElem) -> int:
        """N(a + bφ) = x·x* = a² + ab − b²."""
        return self._r(x.a * x.a + x.a * x.b - x.b * x.b)


class ExactFoldCore:
    """
    Cassini fold engine using the M15 matrix over Z/256Z.

    M15 = [[219, 98],   det(M15) = 219·121 − 98·98 = 26499 − 9604 = 16895 ≡ 219 (mod 256)
            [98, 121]]

    M15_INV = [[135, 98],   M15 · M15_INV ≡ I (mod 256)
               [98,  37]]

    fold(payload)   : applies M15 to successive byte pairs → folded_bytes + Receipt
    unfold(folded)  : applies M15_INV to recover original bytes exactly
    cassini_witness : verifies F(n+1)·F(n-1) − F(n)² ≡ (−1)^n (mod 256)
                      as a live integrity check on the Fibonacci lattice.
    """

    M15     = ((219, 98), (98, 121))
    M15_INV = ((135, 98), (98,  37))
    M15_DET = (219 * 121 - 98 * 98) % 256   # 219*121=26499, 98*98=9604 → 16895 % 256 = 255

    def __init__(self, ring: ExactRing):
        self.ring = ring

    def fold(self, payload: bytes) -> Tuple[bytearray, Receipt]:
        """
        Transform payload bytes pairwise through M15, emit a canonical receipt.

        For each pair (x, y):
            out[2i]   = (219·x + 98·y)  mod 256
            out[2i+1] = (98·x  + 121·y) mod 256

        A trailing odd byte is promoted via the first row of M15 only.
        """
        n   = len(payload)
        out = bytearray(n)

        # Even pairs
        for i in range(n // 2):
            x = payload[2 * i]
            y = payload[2 * i + 1]
            out[2 * i]     = (self.M15[0][0] * x + self.M15[0][1] * y) % 256
            out[2 * i + 1] = (self.M15[1][0] * x + self.M15[1][1] * y) % 256

        if n & 1:
            # Odd trailing byte: encode into two *appended* output bytes so
            # it does not corrupt the preceding even-pair result.
            # out is bytearray(n); we extend it by 1 to hold the second half.
            x = payload[-1]
            out[-1]  = (self.M15[0][0] * x) % 256
            out      = out + bytearray([(self.M15[1][0] * x) % 256])

        payload_dict = {
            "schema_version": "chronus-fold-v1",
            "oVID_id"       : derive_ovid_id(bytes(out)),
            "folded_sha256" : sha256(bytes(out)),
            "tenant_hash"   : self.ring.tenant_hash,
            "M15_det"       : self.M15_DET,
            "at"            : now_iso8601(),
        }
        rh = sha256(rfc8785(payload_dict))
        rs = sign_ed25519(self.ring.signing_key, rh)
        return out, Receipt(rh, rs, self.ring.key_id)

    def unfold(self, folded: bytes, orig_len: int) -> bytes:
        """
        Exact inverse of fold via M15_INV mod 256.
        Invariant I8: unfold(fold(x)) == x  for all byte strings x.

        Mirrors fold exactly:
          For each complete pair (u, v) → recover (x, y) via M15_INV.
          Trailing odd byte was encoded via the first row of M15 only;
          recover it via M15_INV[0][0] alone.
        """
        n     = orig_len
        pairs = n // 2
        out   = bytearray()

        for i in range(pairs):
            u = folded[2 * i]
            v = folded[2 * i + 1]
            out.append((self.M15_INV[0][0] * u + self.M15_INV[0][1] * v) % 256)
            out.append((self.M15_INV[1][0] * u + self.M15_INV[1][1] * v) % 256)

        if n & 1:
            # Trailing odd byte was encoded into the two appended bytes at
            # folded[pairs*2] and folded[pairs*2+1].  Recover via M15_INV row 0.
            u = folded[pairs * 2]
            v = folded[pairs * 2 + 1]
            out.append((self.M15_INV[0][0] * u + self.M15_INV[0][1] * v) % 256)

        return bytes(out[:orig_len])

    def cassini_witness(self, n: int) -> bool:
        """
        Cassini identity: F(n+1)·F(n-1) − F(n)² = (−1)^n
        Verified in Z/256Z.  Returns True iff the identity holds.
        Invariant I6.
        """
        fn,  fnp1 = fib_fast(n)
        fnm1      = fib_fast(n - 1)[0] if n >= 1 else 0
        lhs = (fnp1 * fnm1 - fn * fn) % 256
        rhs = ((-1) ** n)              % 256
        return lhs == rhs

# TIER 13: OVID + THESEUS
#          Durable Object Storage (IBM COS), Immutable Chain & Indexing
# =============================================================================
#
# OVID (Object Versioning & Immutable Derivation):
#   Every write produces a content-addressed OVID ID, a canonical receipt whose
#   hash is sha256(rfc8785(payload)), and a chain head update that makes the
#   lineage append-only.  Receipts are themselves stored as COS objects so the
#   proof layer is durable by the same mechanism as the data.
#
# THESEUS (Tensor History & Exact Schema for Enterprise Update System):
#   Query index over OVID receipts; reconstruct any historical object by ID.
#   Named for the ship whose planks were replaced one by one — every piece of
#   the system can change while the identity (the receipt chain) is preserved.
# =============================================================================

class OVIDService:
    """
    In-memory simulation of IBM Cloud Object Storage + receipt chain.
    In production: each `store_db` write is a COS PutObject with
    if-none-match="*" for idempotency; each `receipts_db` write is a COS
    PutObject to the receipts bucket; `chain_heads` uses a CAS-protected
    head object (compare-and-swap steal on conflict).
    """

    def __init__(self):
        self.store_db:     Dict[str, bytes]         = {}   # COS data bucket
        self.receipts_db:  Dict[str, StoredReceipt] = {}   # COS receipts bucket
        self.chain_heads:  Dict[str, dict]           = {}   # per-tenant head pointer
        self.global_index: Dict[str, str]            = {}   # receipt_hash → tenant_hash

    def store(
        self,
        tenant_hash: str,
        staged_bytes: bytes,
        fold_kind:    str,
        params:       dict
    ) -> dict:
        """
        Atomically:
          1. Write staged_bytes to the data bucket under a content-keyed path.
          2. Build a canonical receipt payload and hash it via RFC 8785.
          3. Sign the receipt hash with the tenant's current Ed25519 key.
          4. Write the StoredReceipt to the receipts bucket.
          5. Advance the per-tenant chain head (CAS-protected in production).
          6. Return {oVID_id, receipt, verify_url}.
        """
        ov      = derive_ovid_id(staged_bytes)
        fsha    = sha256(staged_bytes)
        cos_key = f"{tenant_hash}/{fold_kind}/{ov}"
        self.store_db[cos_key] = staged_bytes

        head = self.chain_heads.get(tenant_hash)
        pos  = (head["position"] + 1) if head else 1

        payload = {
            "schema_version"  : SCHEMA_VERSION,
            "op"              : "ovid.store",
            "oVID_id"         : ov,
            "tenant_hash"     : tenant_hash,
            "fold_kind"       : fold_kind,
            "source_sha256"   : params.get("source_sha256", sha256(staged_bytes)),
            "folded_sha256"   : fsha,
            "chain_position"  : pos,
            "parent_receipt"  : head["receipt_hash"] if head else None,
            "at"              : now_iso8601(),
        }
        rh = sha256(rfc8785(payload))
        rs = sign_ed25519("mock_signing_key", rh)

        stored = StoredReceipt(
            receipt_hash   = rh,
            tenant_hash    = tenant_hash,
            schema_version = SCHEMA_VERSION,
            key_id         = "kid_1",
            payload        = payload,
            receipt_sig    = rs,
            parent_receipt = head["receipt_hash"] if head else None,
            chain_position = pos,
            at             = payload["at"],
        )
        self.receipts_db[rh] = stored
        self.chain_heads[tenant_hash] = {
            "position":     pos,
            "receipt_hash": rh,
            "oVID_id":      ov,
        }
        self.global_index[rh] = tenant_hash

        return {
            "oVID_id"    : ov,
            "receipt"    : Receipt(rh, rs, "kid_1"),
            "verify_url" : PUBLIC_VERIFY_BASE + rh,
        }

    def retrieve(self, tenant_hash: str, ov: str) -> bytes:
        """
        Retrieve stored bytes by OVID ID.
        Invariant I11: retrieve(store(x)) returns byte-identical content.
        """
        for key, val in self.store_db.items():
            if key.startswith(tenant_hash) and key.endswith(ov):
                return val
        raise KeyError(f"Object {ov} not found for tenant {tenant_hash}")


class TheseusService:
    """
    Query and reconstruction layer over OVID receipts.
    query()       : returns all receipt hashes for a tenant matching a predicate.
    reconstruct() : retrieves the stored bytes for a given OVID ID.
    """

    def __init__(self, ovid: OVIDService):
        self.ovid = ovid

    def query(self, tenant_hash: str, q: dict) -> List[str]:
        """
        Returns all receipt hashes for the tenant.
        In production: inverted index on fold_kind, timestamp range, oVID_id prefix.
        """
        return [
            rh for rh, sr in self.ovid.receipts_db.items()
            if sr.tenant_hash == tenant_hash
        ]

    def reconstruct(self, tenant_hash: str, ov: str) -> bytes:
        """Reconstruct object bytes from durable COS store by OVID ID."""
        return self.ovid.retrieve(tenant_hash, ov)

class PublicVerifyService:
    """
    Stateless HTTP handler simulating GET /verify/{receipt_hash}.
    Checks:
      1. Receipt exists in the receipts bucket.
      2. sha256(rfc8785(stored.payload)) == receipt_hash         [I2]
      3. Ed25519 signature is structurally valid                 [I3]
      4. Parent receipt exists and its hash is consistent        [I5]
    Returns HTTP 200 in all cases; {valid: false, reason: ...} on failure
    so that the endpoint never reveals internal error codes.
    """

    @staticmethod
    def handle_get_verify(
        receipt_hash: str,
        ovid_service: OVIDService
    ) -> Tuple[int, dict]:

        stored = ovid_service.receipts_db.get(receipt_hash)
        if not stored:
            return 200, {"valid": False, "reason": "receipt_not_found"}

        # I2: canonical hash check
        recomputed = sha256(rfc8785(stored.payload))
        if recomputed != receipt_hash:
            return 200, {"valid": False, "reason": "payload_hash_mismatch"}

        # I3: signature structure check
        if not verify_ed25519(b"mock_pub", recomputed, stored.receipt_sig):
            return 200, {"valid": False, "reason": "signature_invalid"}

        # I5: parent chain check
        if stored.parent_receipt:
            parent = ovid_service.receipts_db.get(stored.parent_receipt)
            if not parent:
                return 200, {"valid": False, "reason": "parent_missing"}
            ph = sha256(rfc8785(parent.payload))
            if ph != stored.parent_receipt:
                return 200, {"valid": False, "reason": "parent_payload_hash_mismatch"}

        return 200, {"valid": True, "reason": "ok"}
